int bytes = 0;

temp = p[j];
p[j] = p[j + 1];
p[j + 1] = temp;

swaps++;
bytes += 3 * sizeof(struct Parcel);

}

#include <stdio.h>

/* Each bit of one byte stores the ON/OFF state of one device. */
enum { LIGHT = 0, FAN = 1, AC = 2, HEATER = 3, PUMP = 4 };

unsigned char switch_on(unsigned char reg, int dev)
{
return reg | (1u << dev);
}

unsigned char switch_off(unsigned char reg, int dev)
{
return reg & ~(1u << dev);
}

unsigned char toggle(unsigned char reg, int dev)
{
return reg ^ (1u << dev);
}

int is_on(unsigned char reg, int dev)
{
return (reg >> dev) & 1u;
}

/* Counts how many devices are ON */
int count_on(unsigned char reg)
{
int count = 0;
while (reg) {
count += reg & 1u;
reg >>= 1;
}
return count;
}

void show(unsigned char reg)
{
for (int bit = 7; bit >= 0; bit--) putchar(((reg >> bit) & 1u) ? '1' : '0');
putchar('\n');
}

int main(void)
{
unsigned char reg = 0;
reg = switch_on(reg, LIGHT);
reg = switch_on(reg, AC);
show(reg);
reg = toggle(reg, FAN);
reg = switch_off(reg, LIGHT);
show(reg);
printf("AC is %s, PUMP is %s\n", is_on(reg, AC) ? "ON" : "OFF",
is_on(reg, PUMP) ? "ON" : "OFF");
return 0;
}

/* Grade rules: 90-100 S, 80-89 A, 70-79 B, 60-69 C, 50-59 D, below 50 F, invalid '?' */
char grade(int marks)
{
if (marks < 0 || marks > 100) return '?';

switch (marks / 10) {
case 10:
case 9: return 'S';
case 8: return 'A';
case 7: return 'B';
case 6: return 'C';
case 5: return 'D';
default: return 'F';
}
}

void loanSchedule(double balance, double rate, double emi, int print_table) {
int month = 0;
double total_interest = 0.0;

if (emi <= balance * rate) {
printf("Error: EMI (%.2f) must exceed first month's interest (%.2f).\n",
emi, balance * rate);
return;
}

while (balance > 1e-6) {
month++;
if (month > 360) {
printf("Stopped: Loan exceeds 360-month safety limit (unaffordable EMI).\n");
return;
}

double interest = balance * rate;
total_interest += interest;
balance += interest;

/* Last month: pay only remaining balance if smaller than fixed EMI */
double paid = (balance < emi) ? balance : emi;
balance -= paid;

if (print_table) {
printf("Month %2d | Interest: %7.2f | Paid: %7.2f | Balance: %8.2f\n",
month, interest, paid, balance);
}
}

printf("EMI: %7.2f -> Months: %3d | Total Interest: %8.2f\n",
emi, month, total_interest);
}

int main(void) {
printf("--- 1. Hand-Verified Schedule (Rs 10,000 @ 2%%, EMI Rs 2,000) ---\n");

double emis[] = {2000.0, 1000.0, 500.0, 250.0, 205.0, 200.0};

unsigned long checks = 0;

/* Original version: tests all divisors 2, 3, 4, ... */
int is_prime_orig(unsigned long n) {
checks = 0;
if (n < 2) return 0;
for (unsigned long i = 2; i * i <= n; i++) {
checks++;
if (n % i == 0) return 0;
}
return 1;
}

/* Redesigned version: checks 2 once, then tests only odd divisors 3, 5, 7, ... */
int is_prime_fast(unsigned long n) {
checks = 0;
if (n < 2) return 0;
if (n == 2) return 1;
checks++;
if (n % 2 == 0) return 0;

for (unsigned long i = 3; i <= n / i; i += 2) {
checks++;
if (n % i == 0) return 0;
}
return 1;
}

int main(void) {
unsigned long tickets[] = {2, 97, 100, 7919, 1000003};

printf("%-10s | %-18s | %-18s\n", "Ticket", "Original (Res/Div)", "Redesigned (Res/Div)");
printf("-----------------------------------------------------\n");
for (int k = 0; k < 5; k++) {
int r1 = is_prime_orig(tickets[k]);
unsigned long c1 = checks;
int r2 = is_prime_fast(tickets[k]);
unsigned long c2 = checks;

printf("%-10lu | %-9s (%3lu) | %-9s (%3lu)\n",
tickets[k], r1 ? "prime" : "not prime", c1,
r2 ? "prime" : "not prime", c2);
}
return 0;
}
